# The Way Agency — robots.txt

# Content Signals (https://contentsignal.org) declare per-use-case permissions:
#   search:   indexing for traditional search results (Google, Bing, etc.) — yes
#   ai-input: real-time grounding by AI engines (ChatGPT, Claude, Perplexity, Gemini,
#             Google AI Overviews) for answering user questions — yes
#   ai-train: incorporating our content into AI model training datasets — no
User-agent: *
Content-Signal: search=yes, ai-input=yes, ai-train=no
Allow: /

# Sitemap
Sitemap: https://www.thewayagency.com/sitemap.xml

# Disallow build artifacts and internal paths
Disallow: /build/
Disallow: /scripts/
Disallow: /data/

# Admin is auth-gated (returns 401) — block it so Google stops trying.
# /portal/, /partner/, /intake/ rely on noindex meta only — DON'T disallow,
# or Google can't see the noindex tag and may keep stale entries.
Disallow: /admin/

# Allow CSS/JS so Google can render pages
Allow: /src/assets/
Allow: /src/css/
Allow: /src/js/
